  9. PYTHON-СКРИПТ ДЛЯ ПОЛУЧЕНИЯ HLS
═══════════════════════════════════════════════════════════════════════════════════════

  import requests, base64

  def caesar_shift(text, shift=-8):
      result = []
      for char in text:
          if char.isalpha():
              base = 65 if char.isupper() else 97
              result.append(chr((ord(char) - base + shift) % 26 + base))
          else:
              result.append(char)
      return ''.join(result)

  def decode_kodik_url(encoded):
      caesar = caesar_shift(encoded, -8)
      padded = caesar + '=' * (4 - len(caesar) % 4)
      decoded = base64.b64decode(padded).decode('utf-8')
      return 'https:' + decoded if decoded.startswith('//') else decoded

  # Получаем ссылки через /ftor
  payload = {
      "d": "v4.astar.bz",
      "d_sign": "...", "pd": "kodikplayer.com", "pd_sign": "...",
      "ref": "https://v4.astar.bz/", "ref_sign": "...",
      "bad_user": "false", "hash": "HASH", "id": "ID",
      "type": "video", "translation_id": "869", "country": "CN",
  }
  r = requests.post("https://kodikplayer.com/ftor", data=payload, headers={
      'Referer': 'https://kodikplayer.com/video/{id}/{hash}/720p',
      'X-Requested-With': 'XMLHttpRequest',
  })
  data = r.json()
  for quality in ['360', '480', '720']:
      url = decode_kodik_url(data['links'][quality][0]['src'])
      print(f"{quality}p: {url}")

import requests
import re

def get_anidub_hls(anime_url):
    """
    Получает HLS-ссылки с anidub.com

    Args:
        anime_url: URL страницы аниме (например: https://anidub.com/12410-...html)

    Returns:
        dict: {quality: hls_url, ...}
    """
    headers = {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
        'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8',
        'Accept-Language': 'ru-RU,ru;q=0.9',
    }

    # Шаг 1: Получить страницу аниме
    r = requests.get(anime_url, headers=headers, timeout=20)
    r.raise_for_status()

    # Шаг 2: Извлечь URL iframe плеера
    iframe_match = re.search(r'<iframe[^>]+src=["']([^"']+)["']', r.text)
    if not iframe_match:
        raise ValueError("Player iframe not found")

    player_url = iframe_match.group(1)
    if player_url.startswith('//'):
        player_url = 'https:' + player_url

    # Шаг 3: Загрузить плеер
    player_headers = headers.copy()
    player_headers['Referer'] = anime_url

    r2 = requests.get(player_url, headers=player_headers, timeout=20)
    r2.raise_for_status()

    # Шаг 4: Извлечь vsource.src
    vsource_match = re.search(r'<source[^>]+src=["']([^"']+)["']', r2.text)
    if not vsource_match:
        raise ValueError("vsource not found")

    vid_path = vsource_match.group(1)  # vid.php?v=/s12/...

    # Шаг 5: Сформировать полный URL vid.php
    from urllib.parse import urljoin
    vid_url = urljoin(player_url, vid_path)

    # Шаг 6: Получить master M3U8
    r3 = requests.get(vid_url, headers=player_headers, timeout=20)
    r3.raise_for_status()

    # Шаг 7: Распарсить master M3U8
    streams = {}
    stream_pattern = r'#EXT-X-STREAM-INF:([^\n]+)\n([^\n]+)'
    for info, url in re.findall(stream_pattern, r3.text):
        # Извлечь RESOLUTION
        res_match = re.search(r'RESOLUTION=(\d+x\d+)', info)
        resolution = res_match.group(1) if res_match else 'unknown'

        # Извлечь BANDWIDTH
        bw_match = re.search(r'BANDWIDTH=(\d+)', info)
        bandwidth = int(bw_match.group(1)) if bw_match else 0

        streams[resolution] = {
            'url': url,
            'bandwidth': bandwidth,
            'info': info
        }

    return streams

import requests
import re
from urllib.parse import urljoin

def download_ts_segments(chunk_m3u8_url, output_dir='./ts_segments'):
    """
    Скачивает все .ts сегменты из HLS chunk m3u8

    Args:
        chunk_m3u8_url: URL chunk m3u8 (например: https://cdn1.anivids.link/.../chunk.m3u8)
        output_dir: директория для сохранения
    """
    import os
    os.makedirs(output_dir, exist_ok=True)

    headers = {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
        'Referer': 'https://player26.ladonyvesna2005.info/',
    }

    # Скачать chunk m3u8
    r = requests.get(chunk_m3u8_url, headers=headers, timeout=20)
    r.raise_for_status()

    # Извлечь сегменты
    segments = re.findall(r'^(?!#)([^\n]+)', r.text, re.MULTILINE)

    base_url = chunk_m3u8_url.rsplit('/', 1)[0] + '/'

    downloaded = []
    for i, seg in enumerate(segments):
        if not seg.strip():
            continue

        seg_url = urljoin(base_url, seg)
        seg_name = seg.split('?')[0]  # n_0_0_0.ts

        r2 = requests.get(seg_url, headers=headers, timeout=30)
        if r2.status_code == 200:
            filepath = os.path.join(output_dir, f'{i:04d}_{seg_name}')
            with open(filepath, 'wb') as f:
                f.write(r2.content)
            downloaded.append(filepath)
            print(f"Downloaded: {seg_name} ({len(r2.content)} bytes)")
        else:
            print(f"Failed: {seg_name} ({r2.status_code})")

    return downloaded

import subprocess
import os

def ts_to_mp4(ts_files, output_file='output.mp4'):
    """
    Конкатенирует .ts файлы в MP4 через ffmpeg

    Args:
        ts_files: список путей к .ts файлам (отсортированных)
        output_file: имя выходного MP4
    """
    # Создать файл списка для ffmpeg
    list_file = 'ts_list.txt'
    with open(list_file, 'w') as f:
        for ts in ts_files:
            f.write(f"file '{ts}'\n")

    # Запустить ffmpeg
    cmd = [
        'ffmpeg',
        '-f', 'concat',
        '-safe', '0',
        '-i', list_file,
        '-c', 'copy',
        '-bsf:a', 'aac_adtstoasc',
        output_file
    ]

    subprocess.run(cmd, check=True)
    os.remove(list_file)
    print(f"Saved: {output_file}")


# Альтернатива без ffmpeg (чистый Python) — медленнее:
def ts_to_mp4_python(ts_files, output_file='output.mp4'):
    """Простая конкатенация .ts файлов (без перекодировки)"""
    with open(output_file, 'wb') as out:
        for ts in ts_files:
            with open(ts, 'rb') as f:
                out.write(f.read())
    print(f"Saved: {output_file}")
